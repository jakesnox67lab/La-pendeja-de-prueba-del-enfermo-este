local function hasExploitFuncs()
    return type(getrawmetatable) == "function"
       and type(setreadonly) == "function"
       and type(getnamecallmethod) == "function"

local function getDistanceBetweenPlayers(p1, p2)
    if not p1 or not p2 then return math.huge end
    local c1, c2 = p1.Character, p2.Character
    if not c1 or not c2 then return math.huge end
    local r1 = c1:FindFirstChild("HumanoidRootPart") or c1:FindFirstChild("Torso") or c1:FindFirstChild("UpperTorso")
    local r2 = c2:FindFirstChild("HumanoidRootPart") or c2:FindFirstChild("Torso") or c2:FindFirstChild("UpperTorso")
    if not r1 or not r2 then return math.huge end
    return (r1.Position - r2.Position).Magnitude
end

local function getTargetPlayerFromArgs(...)
    for i = 1, select("#", ...) do
        local arg = select(i, ...)
        if type(arg) == "string" then
            local t = Players:FindFirstChild(arg)
            if t and t ~= player then return t end
        elseif type(arg) == "table" and arg.Name then
            local t = Players:FindFirstChild(arg.Name)
            if t and t ~= player then return t end
        elseif type(arg) == "Instance" and arg:IsA("Player") and arg ~= player then
            return arg
        end
    end
    return nil
end

local function isDamageEvent(name)
    local ln = name:lower()
    for _, d in ipairs(DAMAGE_EVENTS) do
        if ln == d:lower() or ln:find(d:lower(), 1, true) then return true end
    end
    return false
end

local function isIgnoredEvent(name)
    local ln = name:lower()
    for _, ig in ipairs(IGNORED_EVENTS) do
        if ln == ig:lower() or ln:find(ig:lower(), 1, true) then return true end
    end
    return false
end

-- ==================== REPEAT EVENT (usa función original) ====================
local function repeatEvent(remote, originalFire, ...)
    local targetObj = getTargetPlayerFromArgs(...)
    if targetObj then
        if getDistanceBetweenPlayers(player, targetObj) > MAX_REPEAT_DISTANCE then
            return false
        end
    end

    local now = tick()
    local rname = remote.Name
    local last = lastFireTime[rname] or 0
    if now - last < 0.05 then return false end
    lastFireTime[rname] = now

    for i = 1, REPEAT_AMOUNT do
        pcall(function()
            if originalFire then
                originalFire(remote, ...)
            elseif remote:IsA("RemoteEvent") then
                remote.FireServer(remote, ...)
            elseif remote:IsA("RemoteFunction") then
                remote.InvokeServer(remote, ...)
            end
        end)
        if i % 10 == 0 then task.wait(0.01) end
    end
    return true
end

local hookedRemotes = {}
local function findAndHookDamageRemotes()
    local function hookRemote(remote)
        if hookedRemotes[remote] then return end
        if not (remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction")) then return end

        local rname = remote.Name
        if isIgnoredEvent(rname) then return end

        if isDamageEvent(rname) or rname:lower():find("damage", 1, true) or rname:lower():find("hit", 1, true) then
            local methodName = remote:IsA("RemoteEvent") and "FireServer" or "InvokeServer"
            local oldFn = remote[methodName]

            local applied = pcall(function()
                remote[methodName] = function(self, ...)
                    if damageRepeaterEnabled then
                        local ok2 = pcall(function()
                            repeatEvent(self, oldFn, ...)
                        end)
                        if ok2 then return end
                    end
                    return oldFn(self, ...)
                end
            end)

            if applied then
                hookedRemotes[remote] = true
                print("[HOOK] " .. rname .. " via method override")
            end
        end
    end

    pcall(function()
        for _, r in pairs(ReplicatedStorage:GetDescendants()) do hookRemote(r) end
    end)
    pcall(function()
        for _, r in pairs(player.PlayerGui:GetDescendants()) do hookRemote(r) end
    end)
    pcall(function()
        for _, r in pairs(Workspace:GetDescendants()) do hookRemote(r) end
    end)
end

local mt, oldNamecall = nil, nil
local metatableHooked = false

local function hookMetatable()
    if metatableHooked then return false end
    if not hasExploitFuncs() then
        print("[WARN] Executor sin funciones de metatable (getrawmetatable/setreadonly)")
        return false
    end

    local success = pcall(function()
        mt = getrawmetatable(game)
        oldNamecall = mt.__namecall
        setreadonly(mt, false)

        mt.__namecall = function(self, ...)
            local method = getnamecallmethod()
            local ok2, name = pcall(function() return self.Name end)
            name = ok2 and name or ""

            if damageRepeaterEnabled and (method == "FireServer" or method == "InvokeServer") then
                if isDamageEvent(name) and not isIgnoredEvent(name) then
                    local targetObj = getTargetPlayerFromArgs(...)
                    if targetObj then
                        if getDistanceBetweenPlayers(player, targetObj) <= MAX_REPEAT_DISTANCE then
                            for i = 1, REPEAT_AMOUNT do
                                oldNamecall(self, ...)
                                if i % 10 == 0 then task.wait(0.005) end
                            end
                            return
                        end
                    end
                end
            end

            return oldNamecall(self, ...)
        end
        metatableHooked = true
    end)
    return success
end

local antiRagdollConnection = nil
local currentCharacterJoints = {}

local function updateJoints(character)
    if not character then return end
    currentCharacterJoints = {}
    for _, joint in pairs(character:GetDescendants()) do
        if joint:IsA("Motor6D") then
            currentCharacterJoints[joint.Name] = {
                Part0 = joint.Part0, Part1 = joint.Part1,
                C0 = joint.C0, C1 = joint.C1
            }
        end
    end
end

local function restoreJoints()
    for name, data in pairs(currentCharacterJoints) do
        if data.Part0 and data.Part0.Parent then
            if data.Part0:FindFirstChild(name) == nil then
                local j = Instance.new("Motor6D")
                j.Name = name
                j.Part0 = data.Part0
                j.Part1 = data.Part1
                j.C0 = data.C0
                j.C1 = data.C1
                j.Parent = data.Part0
            end
        end
    end
end

local function isInRagdoll(hum)
    if not hum then return false end
    local state = hum:GetState()
    return state == Enum.HumanoidStateType.Physics
        or state == Enum.HumanoidStateType.FallingDown
        or hum.PlatformStand
end

local function antiRagdollLoop()
    if not AntiRagdollEnabled then return end
    local ch = player.Character
    if not ch then return end
    local hum = ch:FindFirstChild("Humanoid")
    local rp  = ch:FindFirstChild("HumanoidRootPart")
    if not hum or not rp then return end

    updateJoints(ch)
    if isInRagdoll(hum) then
        restoreJoints()
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Running)
            hum.PlatformStand = false
            hum.Sit = false
            hum.AutoRotate = true
            rp.Velocity = Vector3.new(rp.Velocity.X, 20, rp.Velocity.Z)
        end)
    end
end

local function setupAntiRagdoll(character)
    if antiRagdollConnection then
        antiRagdollConnection:Disconnect()
        antiRagdollConnection = nil
    end
    task.wait(0.5)
    updateJoints(character)
    if AntiRagdollEnabled then
        antiRagdollConnection = RunService.Heartbeat:Connect(antiRagdollLoop)
    end
end

local function toggleAntiRagdoll()
    AntiRagdollEnabled = not AntiRagdollEnabled
    if AntiRagdollEnabled then

        end
    else
        if antiRagdollConnection then

end

local function findPlayerByPartialName(inputText)
    if inputText == "" or inputText:lower() == "todos" or inputText:lower() == "all" then
        return nil, "TODOS"
    end
    local s = inputText:lower():gsub("%s+", "")
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            if p.Name:lower() == s then return p, p.Name end
            if p.DisplayName:lower() == s then return p, p.DisplayName end
        end
    end
    if #s >= 3 then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player then
                if p.Name:lower():sub(1, #s) == s then return p, p.Name end
                if p.DisplayName:lower():sub(1, #s) == s then return p, p.DisplayName end
            end
        end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player then
                if p.Name:lower():find(s, 1, true) then return p, p.Name end
                if p.DisplayName:lower():find(s, 1, true) then return p, p.DisplayName end
            end
        end
    end
    if #s > 0 and #s < 3 then return false, "Minimo 3 letras" end
    return false, "No encontrado"
end

-- ==================== HITBOX ====================
local function resetHitbox(t)
    pcall(function()
        if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
            local rp = t.Character.HumanoidRootPart
            rp.Size = Vector3.new(2, 2, 1)
            rp.Transparency = 1
            rp.Material = Enum.Material.Plastic
            rp.Color = Color3.fromRGB(255, 255, 255)
        end
    end)
end

local function resetAllHitboxes()
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= player then resetHitbox(v) end
    end
end

local function applyHitboxToPlayer(t)
    pcall(function()
        if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
            local rp = t.Character.HumanoidRootPart
            rp.Size = Vector3.new(HITBOX_SIZE, HITBOX_SIZE, HITBOX_SIZE)
            rp.Transparency = HITBOX_TRANSPARENCY
            rp.Color = HITBOX_COLOR
            rp.Material = Enum.Material.SmoothPlastic
            rp.CanCollide = false
        end
    end)
end

local function shouldHitPlayer(p)
    if p == player then return false end
    if not p.Character then return false end
    if not p.Character:FindFirstChild("HumanoidRootPart") then return false end
    return true
end

local function updateHitboxes()
    if not HitboxEnabled then
        resetAllHitboxes()
        return
    end
    if targetPlayer then
        if shouldHitPlayer(targetPlayer) then
            for _, v in pairs(Players:GetPlayers()) do
                if v ~= player and v ~= targetPlayer then resetHitbox(v) end
            end
            applyHitboxToPlayer(targetPlayer)
        else
            resetHitbox(targetPlayer)
        end
    else
        for _, v in pairs(Players:GetPlayers()) do
            if shouldHitPlayer(v) then applyHitboxToPlayer(v) else resetHitbox(v) end
        end
    end
end

local function setHitboxSize(v)
    HITBOX_SIZE = tonumber(v)
    if hitboxSizeLabel then hitboxSizeLabel.Text = "Tamano actual: " .. HITBOX_SIZE end
    if HitboxEnabled then updateHitboxes() end
end

local function setHitboxTransparency(v)
    HITBOX_TRANSPARENCY = tonumber(v)
    if hitboxTransparencyLabel then
        hitboxTransparencyLabel.Text = "Transparencia actual: " .. HITBOX_TRANSPARENCY
    end
    if HitboxEnabled then updateHitboxes() end
end

-- ==================== DRAG ====================
local function MakeDraggableWithHandle(frame, handle)
    local dragging, dragStartPos, frameStartPos = false, nil, nil

    handle.InputBegan:Connect(function(input, gp)
        if gp then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStartPos = input.Position
            frameStartPos = frame.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input, gp)
        if gp then return end
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStartPos
            frame.Position = UDim2.new(
                frameStartPos.X.Scale, frameStartPos.X.Offset + d.X,
                frameStartPos.Y.Scale, frameStartPos.Y.Offset + d.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input, gp)
        if gp then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function updateTargetStatus()
    if not targetStatus then return end
    if targetPlayer then
        targetStatus.Text = "Objetivo: " .. exactTargetName
        targetStatus.TextColor3 = Color3.fromRGB(80, 255, 80)
        if targetBox then targetBox.Text = exactTargetName end
    else
        targetStatus.Text = "Objetivo: TODOS"
        targetStatus.TextColor3 = Color3.fromRGB(120, 200, 255)
    end
end

local function searchAndSetTarget()
    if not targetBox then return end
    local s = targetBox.Text:gsub("%s+", "")
    local found, rname = findPlayerByPartialName(s)
    if found then
        targetPlayer = found
        exactTargetName = rname
        updateTargetStatus()
        if searchResult then
            searchResult.Text = "[OK] Encontrado: " .. exactTargetName
            searchResult.TextColor3 = Color3.fromRGB(80, 255, 80)
        end
        if HitboxEnabled then updateHitboxes() end
    elseif found == nil then
        targetPlayer = nil
        exactTargetName = "TODOS"
        updateTargetStatus()
        if searchResult then
            searchResult.Text = "[OK] Modo: TODOS"
            searchResult.TextColor3 = Color3.fromRGB(120, 200, 255)
        end
        if HitboxEnabled then updateHitboxes() end
    else
        if searchResult then
            searchResult.Text = rname
            searchResult.TextColor3 = Color3.fromRGB(255, 80, 80)
        end
    end
    task.wait(2)
    if searchResult then
        searchResult.Text = "Presiona Enter para buscar"
        searchResult.TextColor3 = Color3.fromRGB(150, 150, 150)
    end
end

local function targetClosest()
    if not searchResult then return end
    local bestD, bestP = math.huge, nil
    local mc = player.Character
    local mr = mc and (mc:FindFirstChild("HumanoidRootPart") or mc:FindFirstChild("Torso") or mc:FindFirstChild("UpperTorso"))
    if not mr then
        searchResult.Text = "[X] No se puede calcular"
        searchResult.TextColor3 = Color3.fromRGB(255, 80, 80)
        task.wait(2)
        searchResult.Text = "Presiona Enter para buscar"
        searchResult.TextColor3 = Color3.fromRGB(150, 150, 150)
        return
    end
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= player then
            pcall(function()
                local ch = v.Character
                if ch then
                    local r = ch:FindFirstChild("HumanoidRootPart") or ch:FindFirstChild("Torso") or ch:FindFirstChild("UpperTorso")
                    if r then
                        local d = (r.Position - mr.Position).Magnitude
                        if d < bestD then bestD = d; bestP = v end
                    end
                end
            end)
        end
    end
    if bestP then
        targetPlayer = bestP
        exactTargetName = bestP.DisplayName .. " (" .. bestP.Name .. ")"
        updateTargetStatus()
        if targetBox then targetBox.Text = bestP.Name end
        searchResult.Text = "[OK] Target: " .. exactTargetName .. " (d:" .. math.floor(bestD) .. ")"
        searchResult.TextColor3 = Color3.fromRGB(80, 255, 80)
        if HitboxEnabled then updateHitboxes() end
        task.wait(2)
        searchResult.Text = "Presiona Enter para buscar"
        searchResult.TextColor3 = Color3.fromRGB(150, 150, 150)
    else
        searchResult.Text = "[X] No hay jugadores cercanos"
        searchResult.TextColor3 = Color3.fromRGB(255, 80, 80)
        task.wait(2)
        searchResult.Text = "Presiona Enter para buscar"
        searchResult.TextColor3 = Color3.fromRGB(150, 150, 150)
    end
end
