local function isDmg(n)
    local l = string.lower(n)
    for _, d in ipairs(DMG_NAMES) do
        local dl = string.lower(d)
        if l == dl or string.find(l, dl, 1, true) then
            return true
        end
    end
    return false
end

local function isIgn(n)
    local l = string.lower(n)
    for _, d in ipairs(IGN_NAMES) do
        local dl = string.lower(d)
        if l == dl or string.find(l, dl, 1, true) then
            return true
        end
    end
    return false
end

local function safeParent(g)
    if type(gethui) == "function" then
        local ok = pcall(function() g.Parent = gethui() end)

    end
    if syn and syn.protect_gui then
        local ok = pcall(function()
            syn.protect_gui(g)

        end)
        if ok then return end

end

local function getDist(a, b)
    if not a or not b then return math.huge end
    local c1 = a.Character
    local c2 = b.Character
    if not c1 or not c2 then return math.huge end
    local r1 = c1:FindFirstChild("HumanoidRootPart") or c1:FindFirstChild("Torso") or c1:FindFirstChild("UpperTorso")
    local r2 = c2:FindFirstChild("HumanoidRootPart") or c2:FindFirstChild("Torso") or c2:FindFirstChild("UpperTorso")
    if not r1 or not r2 then return math.huge end
    return (r1.Position - r2.Position).Magnitude
end

local function getTgtFromArgs(...)
    local args = {...}
    for i = 1, #args do
        local a = args[i]
        if type(a) == "string" then
            local t = Players:FindFirstChild(a)
            if t and t ~= plr then return t end
        elseif type(a) == "table" and a.Name then
            local t = Players:FindFirstChild(a.Name)
            if t and t ~= plr then return t end
        elseif typeof(a) == "Instance" and a:IsA("Player") and a ~= plr then
            return a
        end
    end
end

local function findPlayer(txt)
    if txt == "" or string.lower(txt) == "todos" or string.lower(txt) == "all" then
        return nil, "TODOS"
    end
    local s = string.lower(txt)
    s = string.gsub(s, "%s+", "")
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= plr and string.lower(p.Name) == s then return p, p.Name end
    end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= plr and string.lower(p.DisplayName) == s then return p, p.DisplayName end
    end
    if #s >= 3 then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= plr and string.sub(string.lower(p.Name), 1, #s) == s then return p, p.Name end
        end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= plr then
                if string.find(string.lower(p.Name), s, 1, true) or string.find(string.lower(p.DisplayName), s, 1, true) then
                    return p, p.Name
                end
            end
        end
    end
    if #s > 0 and #s < 3 then return false, "Minimo 3 letras" end
    return false, "No encontrado"
end

local function resetHB(t)
    pcall(function()
        if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
            local rp = t.Character.HumanoidRootPart
            rp.Size = Vector3.new(2, 2, 1)
            rp.Transparency = 1
            rp.Material = Enum.Material.Plastic
            rp.Color = Color3.fromRGB(255, 255, 255)
        end
    end)
end

local function resetAllHB()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= plr then resetHB(p) end
    end
end

local function applyHB(t)
    pcall(function()
        if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
            local rp = t.Character.HumanoidRootPart
            rp.Size = Vector3.new(HB_SIZE, HB_SIZE, HB_SIZE)
            rp.Transparency = HB_TRANS
            rp.Color = Color3.fromRGB(255, 0, 0)
            rp.Material = Enum.Material.SmoothPlastic
            rp.CanCollide = false
        end
    end)
end

local function shouldHit(p)
    return p ~= plr and p.Character and p.Character:FindFirstChild("HumanoidRootPart")
end

local function updateHB()
    if not hitboxOn then
        resetAllHB()
        return
    end
    if target then
        if shouldHit(target) then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= plr and p ~= target then resetHB(p) end
            end
            applyHB(target)
        else
            resetHB(target)
        end
    else
        for _, p in pairs(Players:GetPlayers()) do
            if shouldHit(p) then applyHB(p) else resetHB(p) end
        end
    end
end

local function updateJoints(c)
    if not c then return end
    joints = {}
    for _, j in pairs(c:GetDescendants()) do
        if j:IsA("Motor6D") then
            joints[j.Name] = {Part0 = j.Part0, Part1 = j.Part1, C0 = j.C0, C1 = j.C1}
        end
    end
end

local function ragLoop()
    if not ragOn then return end
    local c = plr.Character
    if not c then return end
    local h = c:FindFirstChild("Humanoid")
    local rp = c:FindFirstChild("HumanoidRootPart")
    if not h or not rp then return end
    updateJoints(c)
    local st = h:GetState()
    if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.FallingDown or h.PlatformStand then
        for n, d in pairs(joints) do
            if d.Part0 and d.Part0.Parent and not d.Part0:FindFirstChild(n) then
                local j = Instance.new("Motor6D")
                j.Name = n
                j.Part0 = d.Part0
                j.Part1 = d.Part1
                j.C0 = d.C0
                j.C1 = d.C1
                j.Parent = d.Part0
            end
        end
        pcall(function()
            h:ChangeState(Enum.HumanoidStateType.Running)
            h.PlatformStand = false
            h.Sit = false
            h.AutoRotate = true
            rp.Velocity = Vector3.new(rp.Velocity.X, 20, rp.Velocity.Z)
        end)
    end
end

local function toggleRag()

        else
            ragBtn.Text = "RAG\nOFF"
            ragBtn.TextColor3 = Color3.fromRGB(255, 80, 80)
        end
    end
end

local function hasMetaFuncs()
    return type(getrawmetatable) == "function"
       and type(setreadonly) == "function"
       and type(getnamecallmethod) == "function"
end

local function repeatCall(orig, self, ...)
    local t = getTgtFromArgs(...)
    if t and getDist(plr, t) > REP_DIST then return false end
    local now = os.clock()
    local last = lastFire[self.Name] or 0
    if now - last < 0.05 then return false end
    lastFire[self.Name] = now
    for i = 1, REP_AMT do
        pcall(function()
            if orig then
                orig(self, ...)
            elseif self:IsA("RemoteEvent") then
                self:FireServer(self, ...)
            elseif self:IsA("RemoteFunction") then
                self:InvokeServer(self, ...)
            end
        end)
        if i % 10 == 0 then task.wait(0.01) end
    end
    return true
end

local function hookRemotes()
    local function check(r)
        if hooked[r] then return end
        if not (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then return end
        local n = r.Name
        if isIgn(n) then return end
        local isHit = isDmg(n) or string.find(string.lower(n), "damage", 1, true) or string.find(string.lower(n), "hit", 1, true)
        if isHit then
            local m = "FireServer"
            if r:IsA("RemoteFunction") then m = "InvokeServer" end
            local orig = r[m]
            local ok = pcall(function()
                r[m] = function(self, ...)
                    if repOn then
                        local repOk = pcall(function()
                            return repeatCall(orig, self, ...)
                        end)
                        if repOk then return end
                    end
                    return orig(self, ...)
                end
            end)
            if ok then hooked[r] = true end
        end
    end
    pcall(function()
        for _, r in pairs(RS:GetDescendants()) do check(r) end
    end)
    pcall(function()
        for _, r in pairs(plr.PlayerGui:GetDescendants()) do check(r) end
    end)
    pcall(function()
        for _, r in pairs(WS:GetDescendants()) do check(r) end
    end)
end

local mt = nil
local origNC = nil
local metaOn = false

local function hookMeta()
    if metaOn then return end
    if not hasMetaFuncs() then return end
    pcall(function()
        mt = getrawmetatable(game)
        origNC = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = function(self, ...)
            local m = getnamecallmethod()
            local ok, n = pcall(function() return self.Name end)
            if not ok then n = "" end
            if repOn and (m == "FireServer" or m == "InvokeServer") and isDmg(n) and not isIgn(n) then
                local t = getTgtFromArgs(...)
                if t and getDist(plr, t) <= REP_DIST then
                    for i = 1, REP_AMT do
                        origNC(self, ...)
                        if i % 10 == 0 then task.wait(0.005) end
                    end
                    return
                end
            end
            return origNC(self, ...)
        end
        metaOn = true
    end)
end

local function toggleRep()
    repOn = not repOn
    if repOn then
        hookRemotes()
        hookMeta()
    else
        if mt and origNC and metaOn then
            pcall(function()
                mt.__namecall = origNC
                setreadonly(mt, true)
            end)
            metaOn = false
        end
    end
    if repBtn then
        if repOn then
            repBtn.Text = "REP\nON"
            repBtn.TextColor3 = Color3.fromRGB(80, 255, 80)
        else
            repBtn.Text = "REP\nOFF"
            repBtn.TextColor3 = Color3.fromRGB(255, 80, 80)
        end
    end
    if repStatLbl then
        if repOn then
            repStatLbl.Text = "ON (" .. REP_AMT .. "x)"
        else
            repStatLbl.Text = "Estado: OFF"
        end
    end
end
